my(N=#P,KA=matrix(7,N,i,j,0),KE=matrix(7,N,i,j,0),KN=vector(N),
VSUM=vector(N),CX=vector(N),CY=vector(N),CZ=vector(N),VPREV=0,VNEW=0);
\\rescale to radius 1
for(k=1,N,P[k]=P[k]/sqrt(sum(j=1,3,P[k][j]^2)));
\\ edges to surrounding points
for(i=1,N,my(k=0);
  for(L=1,#IKN,
   if(IKN[L][1]==i,k++;KA[k,i]=IKN[L][2];KE[k,i]=IKN[L][3]);
   if(IKN[L][2]==i,k++;KA[k,i]=IKN[L][3];KE[k,i]=IKN[L][1]);
   if(IKN[L][3]==i,k++;KA[k,i]=IKN[L][1];KE[k,i]=IKN[L][2]));
  KN[i]=k);

for(i=1,N,
for(L=1,KN[i],my(j=KA[L,i],k=KE[L,i]);
CX[i]=CX[i]-P[j][2]*P[k][3] + P[k][2]*P[j][3];
CY[i]=CY[i]-P[j][3]*P[k][1] + P[k][3]*P[j][1];
CZ[i]=CZ[i]-P[j][1]*P[k][2] + P[k][1]*P[j][2]);

for(it=1,abs(itmax),my(iverbz=0);
 for(ii=1,N,if(it%2,i=ii,i=N-ii+1);
my(CB=sqrt(CX[i]^2+CY[i]^2+CZ[i]^2),PN=[CX[i]/CB,CY[i]/CB,CZ[i]/CB],
VN=PN[1]*CX[i]+PN[2]*CY[i]+PN[3]*CZ[i]);
if(itmax>0,print([it,i,VN,VSUM[i]]));
if(VN>VSUM[i],if(itmax>0,print("Improved at P",i,":",VN-VSUM[i]));iverbz++;

 for(LL=1,KN[M],my(j=KA[LL,M],k=KE[LL,M]);
  CX[M]=CX[M]-P[j][2]*P[k][3] + P[k][2]*P[j][3];
  CY[M]=CY[M]-P[j][3]*P[k][1] + P[k][3]*P[j][1];
  CZ[M]=CZ[M]-P[j][1]*P[k][2] + P[k][1]*P[j][2]
    );
 VSUM[M]=P[M][1]*CX[M]+P[M][2]*CY[M]+P[M][3]*CZ[M])
 ));

         -P[i][2]*(P[j][3]*P[k][1]-P[k][3]*P[j][1])
         -P[i][3]*(P[j][1]*P[k][2]-P[k][1]*P[j][2]));
if(VNEW>VPREV,print("it=",it," ",VNEW/6);VPREV=VNEW,
             print(iverbz," no improvement:",VNEW/6));
if(iverbz<N\3,print("No improvement in iteration ",it);break));

amin=min(amin,vecmin([d1,d2,d3]));amax=max(amax,vecmax([d1,d2,d3])));
